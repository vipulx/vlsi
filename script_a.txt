## 1. Find multiples of 3 (up to n)

```bash
#!/bin/bash
echo "Enter a number:"
read n

echo "Multiples of 3 up to $n are:"
for ((i=1; i<=n; i++))
do
  if [ $((i % 3)) -eq 0 ]; then
    echo $i
  fi
done
```

## 2. Triangle type (Scalene / Isosceles / Equilateral)

```bash
#!/bin/bash
echo "Enter three sides:"
read a b c

if [ $a -eq $b ] && [ $b -eq $c ]; then
  echo "Equilateral Triangle"
elif [ $a -eq $b ] || [ $b -eq $c ] || [ $a -eq $c ]; then
  echo "Isosceles Triangle"
else
  echo "Scalene Triangle"
fi
```

## 3. All arithmetic operations on two numbers

```bash
#!/bin/bash
echo "Enter two numbers:"
read a b

echo "Addition: $((a + b))"
echo "Subtraction: $((a - b))"
echo "Multiplication: $((a * b))"

if [ $b -ne 0 ]; then
  echo "Division: $((a / b))"
else
  echo "Division not possible (division by zero)"
fi
```

## 4. Display Fibonacci series

```bash
#!/bin/bash
echo "Enter number of terms:"
read n

a=0
b=1
echo "Fibonacci series:"
for ((i=0; i<n; i++))
do
  echo -n "$a "
  fn=$((a + b))
  a=$b
  b=$fn
done
echo
```

## 5. Roots of quadratic equation

Equation: **ax² + bx + c = 0**

```bash
#!/bin/bash
echo "Enter a b c:"
read a b c

d=$((b*b - 4*a*c))

if [ $d -gt 0 ]; then
  r1=$(echo "scale=2; (-$b + sqrt($d))/(2*$a)" | bc)
  r2=$(echo "scale=2; (-$b - sqrt($d))/(2*$a)" | bc)
  echo "Roots are real and different: $r1 , $r2"
elif [ $d -eq 0 ]; then
  r=$(echo "scale=2; -$b/(2*$a)" | bc)
  echo "Roots are real and same: $r"
else
  echo "Roots are imaginary"
fi
```

## 6. Descending order using `for` loop

```bash
#!/bin/bash
echo "Enter a number:"
read n

for ((i=n; i>=1; i--))
do
  echo $i
done
```

## 10. Calculate SGPA

```bash
#!/bin/bash
echo "Enter number of subjects:"
read n

total=0
total_credits=0

for ((i=1; i<=n; i++))
do
  echo "Enter grade point for subject $i:"
  read gp
  echo "Enter credits for subject $i:"
  read cr

  total=$((total + gp * cr))
  total_credits=$((total_credits + cr))
done

sgpa=$(echo "scale=2; $total / $total_credits" | bc)
echo "SGPA = $sgpa"
```
